<script>

let data = JSON.parse(localStorage.getItem("data") || "[]");
let people = JSON.parse(localStorage.getItem("people") || "[]");

/* LOGIN */
function login(){

let u=user.value;
let p=pass.value;

if(u==="admin" && p==="1234"){
loginBox.style.display="none";
app.style.display="block";
adminPanel.style.display="block";
title.innerText="Admin";
renderAll();
return;
}

if(u==="besiroglu" && p==="6161"){
loginBox.style.display="none";
app.style.display="block";
reportPanel.style.display="block";
title.innerText="Rapor";
renderAll();
return;
}

alert("Hatalı giriş");
}

/* 🚪 LOGOUT */
function logout(){

loginBox.style.display="block";
app.style.display="none";
adminPanel.style.display="none";
reportPanel.style.display="none";

user.value="";
pass.value="";
}

/* PERSON */
function addPerson(){
people.push({id:Date.now(),name:newName.value});
localStorage.setItem("people",JSON.stringify(people));
newName.value="";
renderAll();
}

/* DELETE PERSON */
function deleteSelectedPerson(){

let name = manageName.value;

people = people.filter(p=>p.name!==name);
data = data.filter(d=>d.name!==name);

localStorage.setItem("people",JSON.stringify(people));
localStorage.setItem("data",JSON.stringify(data));

renderAll();
}

/* ADD */
function add(){

data.push({
id:Date.now(),
name:nameSelect.value,
amount:Number(amount.value),
date:date.value
});

localStorage.setItem("data",JSON.stringify(data));
renderAll();
}

/* DELETE DATA */
function delData(id){
data = data.filter(d=>d.id!==id);
localStorage.setItem("data",JSON.stringify(data));
renderAll();
}

/* RENDER ALL */
function renderAll(){

nameSelect.innerHTML = people.map(p=>
`<option value="${p.name}">${p.name}</option>`
).join("");

manageName.innerHTML = people.map(p=>
`<option value="${p.name}">${p.name}</option>`
).join("");

filterName.innerHTML =
`<option value="all">Tümü</option>` +
people.map(p=>
`<option value="${p.name}">${p.name}</option>`
).join("");

render();
report();
}

/* ADMIN TABLE */
function render(){

list.innerHTML="";

for(let d of data){

let ok = d.amount > 0;

list.innerHTML += `
<tr>
<td>${d.name}</td>
<td>${d.amount}</td>
<td>${d.date}</td>
<td class="${ok?'green':'red'}">
${ok?'Ödedi':'Ödemedi'}
</td>
<td><button class="xbtn" onclick="delData(${d.id})">x</button></td>
</tr>`;
}
}

/* REPORT */
function report(){

let s=new Date(start.value||"1900-01-01");
let e=new Date(end.value||"2999-12-31");

let f=data.filter(d=>{
let x=new Date(d.date);
let nameOk = filterName.value==="all" || d.name===filterName.value;
return x>=s && x<=e && nameOk;
});

let t=f.reduce((a,b)=>a+b.amount,0);

total.innerText="Toplam: "+t;

rlist.innerHTML="";

for(let d of f){
rlist.innerHTML+=`
<tr>
<td>${d.name}</td>
<td>${d.amount}</td>
<td>${d.date}</td>
<td><button class="xbtn" onclick="delData(${d.id})">x</button></td>
</tr>`;
}
}

/* EXCEL */
function excel(){
let ws=XLSX.utils.json_to_sheet(data);
let wb=XLSX.utils.book_new();
XLSX.utils.book_append_sheet(wb,ws,"Data");
XLSX.writeFile(wb,"rapor.xlsx");
}
